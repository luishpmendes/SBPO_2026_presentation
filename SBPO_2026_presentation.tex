\begin{frame}
\frametitle{Do FOTA ao TMDSP}

{\small \textbf{Tarefas} $J$ (ve\'iculos), \textbf{m\'aquinas} $M$ (setores) e
\textbf{per\'iodos} $T$. Se a tarefa $j$ for atendida pela m\'aquina $m$ no
per\'iodo $t$, consome $q_{jmt}$ de capacidade:}

\smallskip

\begin{center}
\resizebox{0.88\textwidth}{!}{\input{tikz/instance_preprocessed.tex}}
\end{center}

\vfill

\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
{\small
\textbf{Restri\c{c}\~oes:} capacidade $c_{mt}$ de cada m\'aquina, vari\'avel no
tempo; limite global $r_{\max}$ por per\'iodo; no m\'aximo $k_{\max}$ tarefas
ativas; demanda $d_{j}$ atendida ao longo de v\'arios per\'iodos.}

\column{0.43\textwidth}
{\small
\textbf{Objetivo original} (mono-objetivo):
\[
\min \sum_{j \in J} C_{j}
\]
\vspace{-0.9em}

tempo total de conclus\~ao.}
\end{columns}

\vspace{0.2em}

Adaptado de Andrade et al. (2019)\par}

%----------------------------------------------------------------------------------------
% 4. POR QUE MULTIOBJETIVO
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Minimizar $\sum_{j} C_{j}$ n\~ao controla o \emph{makespan}}

\begin{columns}[c,onlytextwidth]
\column{0.47\textwidth}

\small
Andrade et al. (2019) observaram que a correla\c{c}\~ao entre os dois
crit\'erios \'e \textbf{fraca}: um escalonamento \'otimo em $\sum_{j} C_{j}$
pode deixar algumas tarefas terminando \textbf{muito depois} das demais.

\medskip

Propomos o \alert{MOTMDSP}, que minimiza \textbf{simultaneamente}:

\smallskip

\begin{center}
\small
$\displaystyle \min\ C_{\max} = \max_{j \in J} C_{j}$
\;\; (\emph{makespan})

\smallskip

$\displaystyle \min \sum_{j \in J} C_{j}$
\;\; (tempo total de conclus\~ao)
\end{center}

\column{0.49\textwidth}
\centering
\begin{tikzpicture}[scale=0.72, font=\small]
\draw[-Stealth, thick] (0,0) -- (7.4,0) node[below left=2pt and 0pt] {$C_{\max}$};
\draw[-Stealth, thick] (0,0) -- (0,4.9) node[above right=0pt and -6pt] {$\sum_{j} C_{j}$};
\draw[ultra thick, mydarkblue1, smooth, tension=0.7]
    plot coordinates {(1.0,4.2) (1.7,3.1) (2.6,2.4) (3.7,2.0) (5.0,1.8) (6.4,1.7)};

    \fill[mydarkblue1] (\x,\y) circle (3pt);
}
\node[mydarkblue1, font=\small\bfseries, anchor=west] at (3.4,3.6)
  {fronteira de Pareto};

\end{tikzpicture}
\end{columns}

\end{frame}

\begin{frame}
\frametitle{Chaves aleat\'orias definem prioridades de escalonamento}

Um cromossomo \'e um vetor com \textbf{uma chave por tarefa}:
\[
\mathbf{v} = (v_1, \ldots, v_{|J|}), \qquad v_j \in [0,\,1).
\]

\textbf{Decodifica\c{c}\~ao:} ordenar as tarefas por valor \textbf{decrescente}
da chave $\Rightarrow$ ordem de prioridade.

\bigskip

\begin{center}
\begin{tikzpicture}[>=Stealth, font=\small, x=1cm, y=1cm]
% vetor original
\node[font=\small, anchor=east] at (-0.52,0) {$\mathbf{v}$:};
\foreach \val [count=\i from 0] in
  {{$0{,}6$}, {$0{,}9$}, {$0{,}8$}, {$0{,}5$}, {$0{,}3$}, {$0{,}2$}} {
    \node[draw, minimum width=0.86cm, minimum height=0.66cm] at (\i*0.88, 0) {\val};
    \node[font=\scriptsize, above=0pt] at (\i*0.88, 0.33)
      {$j_{\the\numexpr\i+1}$};
}
% seta
\draw[-Stealth, ultra thick, mydarkorange1] (4.95,0) -- (6.15,0);
\node[font=\scriptsize, align=center, below=1pt] at (5.55,-0.10)
  {ordem\\decrescente};
% vetor ordenado
\foreach \job/\col/\txt [count=\i from 0] in
  {{$j_2$}/mylightorange1/black, {$j_3$}/mylightblue1/black,
   {$j_1$}/white/black, {$j_4$}/white/black,
   {$j_5$}/white/black, {$j_6$}/white/black} {
    \node[draw, fill=\col, text=\txt, minimum width=0.68cm, minimum height=0.66cm]
        at (6.55+\i*0.72, 0) {\job};
}
\node[font=\scriptsize, above=0pt] at (8.35,0.33) {\textbf{prioridade}};
\end{tikzpicture}
\end{center}

\bigskip

Pequenas perturba\c{c}\~oes nas chaves causam pequenas mudan\c{c}as na ordem, o
que permite explorar o espa\c{c}o de permuta\c{c}\~oes \textbf{sem operadores
especializados}.

\end{frame}

% 12. LEITURA DOS RESULTADOS

\begin{frame}
\frametitle{Como ler esses resultados}

\begin{itemize}\setlength{\itemsep}{10pt}
\item As diferen\c{c}as entre os m\'etodos concentram-se entre a
      \textbf{terceira e a quinta casa decimal} dos indicadores, e as
      distribui\c{c}\~oes por execu\c{c}\~ao \textbf{se sobrep\~oem}.

\item \textbf{N\~ao} realizamos testes de hip\'otese: os resultados devem ser
      lidos como \textbf{desempenho emp\'irico observado}, e n\~ao como
      superioridade estatisticamente significativa.

\item Nenhum algoritmo apresenta degrada\c{c}\~ao \`a medida que o n\'umero de
      tarefas ou de m\'aquinas cresce, sob o or\c{c}amento fixo de tempo.

\item Os par\^ametros dos algoritmos de refer\^encia s\~ao os padr\~oes do
      PaGMO2; os do \alg{NS-BRKGA} v\^em da literatura de BRKGA. As origens
      diferem, ent\~ao \textbf{n\~ao afirmamos uniformidade de calibra\c{c}\~ao}.
\end{itemize}

\end{frame}

\section{Conclusão}

%----------------------------------------------------------------------------------------
% 13. CONCLUSÕES
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Conclus\~oes}

\begin{itemize}\setlength{\itemsep}{10pt}
\item Formulamos o \textbf{MOTMDSP}, extens\~ao biobjetivo do TMDSP que trata
      explicitamente o compromisso entre \emph{makespan} e tempo total de
      conclus\~ao.

\item Projetamos um \textbf{decodificador} que transforma prioridades em
      escalonamentos vi\'aveis e permitiu aplicar o \alg{NS-BRKGA} ---
      e tamb\'em os cinco algoritmos de refer\^encia --- ao problema.

\item Em $96$ inst\^ancias de larga escala, o \alg{NS-BRKGA} obteve as
      \textbf{melhores m\'edias} de $\HVR$ e $\NIGDplus$, com vantagem mais
      n\'itida nas inst\^ancias \texttt{rare}.

\item \textbf{Uso pr\'atico:} em vez de um \'unico escalonamento, o operador
      recebe um \textbf{conjunto de compromissos} entre terminar a campanha
      mais cedo e concluir mais ve\'iculos antes.
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Apoio: par\^ametros do \alg{NS-BRKGA}}

\begin{center}
\small
\begin{tabular}{@{} l l @{}}
\toprule
\textbf{Par\^ametro} & \textbf{Valor} \\
\midrule
Tamanho da popula\c{c}\~ao $|P|$ & $300$ \\
N\'umero de pais $\pi$ (elitistas $\pi_{e}$) & $3$ (com $\pi_{e} = 2$) \\
Fra\c{c}\~ao de elite $p^{e}_{\min}$ -- $p^{e}_{\max}$ & $0{,}10$ -- $0{,}30$ \\
Fun\c{c}\~ao de vi\'es $\Phi(r)$ & $1/\sqrt{r}$ \\
Probabilidade de muta\c{c}\~ao $p_{m}$ ($\eta_{m}$) & $0{,}01$ ($\eta_{m} = 50$) \\
N\'umero de subpopula\c{c}\~oes & $3$ \\
Migra\c{c}\~ao & $200$ gera\c{c}\~oes, $k = 30$ solu\c{c}\~oes \\
\emph{Implicit path-relinking} & a cada $500$ gera\c{c}\~oes \\
\emph{Shaking} & $200$ gera\c{c}\~oes sem melhora \\
\emph{Reset} parcial & $500$ gera\c{c}\~oes sem melhora \\
\bottomrule
\end{tabular}
\end{center}

\end{frame}
