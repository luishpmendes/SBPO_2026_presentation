%----------------------------------------------------------------------------------------
% 3. DO FOTA AO TMDSP
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Do FOTA ao TMDSP}

{\small O FOTA \'e modelado pelo \emph{Time- and Machine-Dependent Scheduling
Problem} (TMDSP), de Andrade et al.\ (2019):}

\smallskip

% Sobrescreve, s\'o neste frame, o \normalsize que o pre\^ambulo fixa
% para o corpo das listas.
\setbeamertemplate{itemize/enumerate body begin}{\footnotesize}

\begin{columns}[T,onlytextwidth]
\column{0.46\textwidth}
\footnotesize
\textbf{O FOTA vira escalonamento}
\begin{itemize}\setlength{\itemsep}{4pt}
\item<1-> $J$ --- \textbf{tarefas}: os ve\'iculos a atualizar.
\item<1-> $M$ --- \textbf{m\'aquinas}: os setores da rede.
\item<1-> $T$ --- \textbf{per\'iodos}: os intervalos do horizonte.
\end{itemize}

\column{0.50\textwidth}
\footnotesize
\textbf{Regras de um escalonamento vi\'avel}
\begin{itemize}\setlength{\itemsep}{4pt}
\item<2-> Cada tarefa usa \textbf{uma m\'aquina por per\'iodo}, no m\'aximo;
\item<2-> \'e \textbf{preempt\'avel entre per\'iodos} e fica \emph{ativa}
      at\'e atender toda a sua demanda;
\item<2-> $C_{j}$ \'e o \textbf{\'ultimo per\'iodo ativo} de $j$.
\end{itemize}
\end{columns}

\vspace{0.4em}

\onslide<3->{\takeaway{Objetivo original, mono-objetivo:
$\displaystyle \min \sum_{j \in J} C_{j}$ --- o tempo total de conclus\~ao.}}

\end{frame}

%----------------------------------------------------------------------------------------
% 4. POR QUE MULTIOBJETIVO
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Minimizar $\sum_{j} C_{j}$ n\~ao controla o \emph{makespan}}

\begin{columns}[c,onlytextwidth]
\column{0.47\textwidth}

\small
A correla\c{c}\~ao entre os dois crit\'erios \'e \textbf{fraca}: um
escalonamento \'otimo em $\sum_{j} C_{j}$ pode deixar algumas tarefas
terminando \textbf{muito depois} das demais.

\medskip

Propomos o \alert{MOTMDSP}, que minimiza \textbf{simultaneamente}:

\smallskip

\begin{center}
\small
$\displaystyle \min\ C_{\max} = \max_{j \in J} C_{j}$
\;\; (\emph{makespan})

\smallskip

$\displaystyle \min \sum_{j \in J} C_{j}$
\;\; (tempo total de conclus\~ao)
\end{center}

\smallskip

{\scriptsize\color{black!55} Observado por Andrade et al.\ (2019).}

\column{0.49\textwidth}
\centering
\begin{tikzpicture}[scale=0.72, font=\small]
\draw[-Stealth, thick] (0,0) -- (7.4,0) node[below left=2pt and 0pt] {$C_{\max}$};

\end{tikzpicture}
\end{columns}

\end{frame}

% NOTAÇÃO: CONJUNTOS E PARÂMETROS (Tabela 1 do artigo)

\begin{frame}
\frametitle{Nota\c{c}\~ao: conjuntos e par\^ametros}

\begin{center}
\footnotesize
\begin{tabular}{@{} l l @{}}
\toprule
\multicolumn{2}{@{}l}{\textbf{Conjuntos}} \\
\midrule
$J$         & Tarefas (ve\'iculos a serem atualizados) \\
$M$         & M\'aquinas (setores de torres celulares) \\
$T$         & Per\'iodos-base em um horizonte (indexados por $t$) \\
\midrule
\multicolumn{2}{@{}l}{\textbf{Par\^ametros}} \\
\midrule
$c_{mt}$    & Capacidade da m\'aquina $m$ no per\'iodo $t$ (muu) \\
$\eta_{mt}$ & Taxa de produ\c{c}\~ao da m\'aquina $m$ no per\'iodo $t$ (pu por muu) \\
$q_{jmt}$   & Consumo de capacidade da tarefa $j$ em $m$ no per\'iodo $t$ (muu) \\
$d_{j}$     & Demanda total da tarefa $j$ (pu) \\
$r_{\max}$  & Limite global de capacidade por per\'iodo (pu) \\
$r_{\min}$  & Consumo m\'inimo de recursos por tarefa (pu) \\
$k_{\max}$  & N\'umero m\'aximo de tarefas ativas em um per\'iodo \\
\bottomrule
\end{tabular}
\end{center}

\vspace{0.2em}

{\scriptsize \textbf{Horizonte c\'iclico:} estendemos $T$ para
$\Text = \{0, 1, \ldots, t_{\max}\}$ e mapeamos cada $t \in \Text$ de volta ao
per\'iodo-base $\hat{t} = t \bmod |T|$; da\'i as vers\~oes $c_{m\hat{t}}$,
$\eta_{m\hat{t}}$, $q_{jm\hat{t}}$ usadas no modelo.
\; muu: \emph{machine utilization units}; pu: \emph{production units}.}

\end{frame}

%----------------------------------------------------------------------------------------
% VARIÁVEIS DE DECISÃO
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Vari\'aveis de decis\~ao}

\begin{center}
\small
\begin{tabular}{@{} l l @{}}
\toprule
\textbf{Vari\'avel} & \textbf{Significado} \\
\midrule
$x_{jmt} \in \{0, 1\}$
  & $1$ se a tarefa $j$ \'e atribu\'ida \`a m\'aquina $m$ no per\'iodo $t$ \\
$y_{jmt} \in \mathbb{R}_{\ge 0}$
  & capacidade da m\'aquina $m$ usada por $j$ no per\'iodo $t$ (muu) \\
$s_{jt} \in \{0, 1\}$
  & $1$ se a tarefa $j$ \textbf{inicia} no per\'iodo $t$ \\
$a_{jt} \in \{0, 1\}$
  & $1$ se a tarefa $j$ est\'a \textbf{ativa} em $t$ (escalonada ou em execu\c{c}\~ao) \\
$f_{jt} \in \{0, 1\}$
  & $1$ se a tarefa $j$ \textbf{termina} em $t$ (indicador de \emph{pseudo}-t\'ermino) \\
$C_{j} \in \mathbb{N}$
  & tempo de conclus\~ao de $j$: o \'indice do per\'iodo em que $j$ termina \\
$C_{\max} \in \mathbb{N}$
  & \emph{makespan}: o maior tempo de conclus\~ao entre todas as tarefas \\
\bottomrule
\end{tabular}
\end{center}

\smallskip

{\footnotesize Para $t \in \Text\setminus\{0\}$, $t'(j, t)$ denota o
\textbf{per\'iodo predecessor imediato} de $t$ no horizonte c\'iclico, usado
para propagar a atividade da tarefa.}

\srcnote{O MOTMDSP \'e formulado como um programa inteiro misto (MIP) que segue
de perto o modelo mono-objetivo de Andrade et al. (2019)}

\end{frame}

%----------------------------------------------------------------------------------------
% FORMULAÇÃO I: OBJETIVOS E LÓGICA DE ATRIBUIÇÃO -- (1)--(6)
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Formula\c{c}\~ao I: objetivos e l\'ogica de atribui\c{c}\~ao}

{\small
\begin{alignat}{1}
\min \ & \sum_{j \in J}{C_{j}} \tag{1} \\
\min \ & C_{\max} \tag{2}
\end{alignat}}

\vspace{-1.1em}

{\small
\begin{alignat}{2}
\text{sujeito a} \quad
&C_{j} \ge t\,a_{jt},\quad&&\forall j \in J,\ \forall t \in \Text \tag{3} \\
&\sum_{m \in M}{x_{jmt}} \le 1,\quad&&\forall j \in J,\ \forall t \in \Text \tag{4} \\
&\sum_{t \in \Text}{s_{jt}} = 1,\quad&&\forall j \in J \tag{5} \\
&a_{jt} \ge \frac{1}{|M|}\sum_{m \in M}{x_{jmt}},\quad&&\forall j \in J,\ \forall t \in \Text \tag{6}
\end{alignat}}

\vfill

{\scriptsize
(3)--(6) definem os tempos de conclus\~ao e a l\'ogica de atribui\c{c}\~ao e
in\'icio: $C_{j}$ \'e o \'ultimo per\'iodo ativo, cada tarefa \'e atribu\'ida a
no m\'aximo uma m\'aquina por per\'iodo e inicia exatamente uma vez.}

\end{frame}

%----------------------------------------------------------------------------------------
% FORMULAÇÃO II: ATIVIDADE, TÉRMINO E CONSUMO -- (7)--(12)
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Formula\c{c}\~ao II: atividade, t\'ermino e consumo}

{\small
\begin{alignat}{2}
&a_{j0} = s_{j0},\quad&&\forall j \in J \tag{7} \\
&a_{jt} \le s_{jt} + a_{j t'(j,t)},\quad&&\forall j \in J,\ \forall t \in \Text\setminus\{0\} \tag{8} \\
&f_{j t_{\max}} = a_{j t_{\max}},\quad&&\forall j \in J \tag{9} \\
&f_{jt} \le 1 - \frac{1}{|\Text|}\sum_{t'>t}{a_{jt'}},\quad&&\forall j \in J,\ \forall t \in \Text\setminus\{t_{\max}\} \tag{10} \\
&y_{jmt} \le q_{jm\hat{t}}\,x_{jmt},\quad&&\forall j \in J,\ \forall m \in M,\ \forall t \in \Text \tag{11} \\
&y_{jmt} \ge q_{jm\hat{t}}\,\big(x_{jmt} - f_{jt}\big),\quad&&\forall j \in J,\ \forall m \in M,\ \forall t \in \Text \tag{12}
\end{alignat}}

\vfill

{\scriptsize
(7)--(8) propagam a atividade ao longo do horizonte, sem ``pular'' per\'iodos
dispon\'iveis, at\'e a conclus\~ao. (9)--(10) identificam o per\'iodo final.
(11)--(12) conectam $y_{jmt}$ e $x_{jmt}$: o consumo s\'o \'e permitido quando
a tarefa est\'a atribu\'ida e, exceto no per\'iodo de t\'ermino, imp\~oe-se o
uso integral $q_{jm\hat{t}}$, evitando subutiliza\c{c}\~ao.}

\end{frame}

%----------------------------------------------------------------------------------------
% FORMULAÇÃO III: CAPACIDADES, DEMANDA E MAKESPAN -- (13)--(17)
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Formula\c{c}\~ao III: capacidades, demanda e \emph{makespan}}

{\small
\begin{alignat}{2}
&\sum_{j \in J}{y_{jmt}} \le c_{m\hat{t}},\quad&&\forall m \in M,\ \forall t \in \Text \tag{13} \\
&\sum_{j \in J}\sum_{m \in M}{\eta_{m\hat{t}}\,y_{jmt}} \le r_{\max},\quad&&\forall t \in \Text \tag{14} \\
&\sum_{j \in J}{a_{jt}} \le k_{\max},\quad&&\forall t \in \Text \tag{15} \\
&\sum_{m \in M}\sum_{t \in \Text}{\eta_{m\hat{t}}\,y_{jmt}} = d_{j},\quad&&\forall j \in J \tag{16} \\
&C_{\max} \ge C_{j},\quad&&\forall j \in J \tag{17}
\end{alignat}}

\vspace{-0.4em}

{\scriptsize
(13)--(15) imp\~oem a capacidade de cada m\'aquina, o limite global por
per\'iodo e o m\'aximo de tarefas ativas. (16) garante o atendimento da demanda
$d_{j}$ e (17) define o \emph{makespan} $C_{\max}$. O requisito m\'inimo
$r_{\min}$ \'e tratado no \textbf{pr\'e-processamento}, removendo as tarefas
n\~ao escalon\'aveis.}

\end{frame}

\begin{frame}
\frametitle{Chaves aleat\'orias definem prioridades de escalonamento}

{\small Um cromossomo \'e um vetor com \textbf{uma chave por tarefa}:}
\vspace{-0.5em}
\[
\textstyle \mathbf{v} = (v_1, \ldots, v_{|J|}), \qquad v_j \in [0,\,1).
\]
\vspace{-1.4em}

{\small\textbf{Decodifica\c{c}\~ao:} ordenar as tarefas por valor
\textbf{decrescente} da chave $\Rightarrow$ ordem de prioridade.}

\smallskip

\begin{center}
\begin{tikzpicture}[>=Stealth, font=\small, x=0.86cm, y=0.86cm]
% vetor original
\node[font=\small, anchor=east] at (-0.52,0) {$\mathbf{v}$:};
\foreach \val [count=\i from 0] in
  {{$0{,}6$}, {$0{,}9$}, {$0{,}8$}, {$0{,}5$}, {$0{,}3$}, {$0{,}2$}} {
    \node[draw, minimum width=0.86cm, minimum height=0.66cm] at (\i*0.88, 0) {\val};
    \node[font=\scriptsize, above=0pt] at (\i*0.88, 0.33)
      {$j_{\the\numexpr\i+1}$};
}
% seta
\draw[-Stealth, ultra thick, mydarkorange1] (4.95,0) -- (6.15,0);
\node[font=\scriptsize, align=center, below=1pt] at (5.55,-0.10)
  {ordem\\decrescente};
% vetor ordenado
\foreach \job/\col/\txt [count=\i from 0] in
  {{$j_2$}/mylightorange1/black, {$j_3$}/mylightblue1/black,
   {$j_1$}/white/black, {$j_4$}/white/black,
   {$j_5$}/white/black, {$j_6$}/white/black} {
    \node[draw, fill=\col, text=\txt, minimum width=0.68cm, minimum height=0.66cm]
        at (6.55+\i*0.72, 0) {\job};
}
\node[font=\scriptsize, above=0pt] at (8.35,0.33) {\textbf{prioridade}};
\end{tikzpicture}
\end{center}

\smallskip

{\footnotesize
Nessa ordem, cada tarefa ocupa gulosamente os intervalos m\'aquina--per\'iodo
dispon\'iveis mais cedo. $C_{j}$ \'e o \'ultimo per\'iodo em que a tarefa
esteve ativa; alocadas \emph{todas} as tarefas, o decodificador devolve
$\mathbf{F} = (C_{\max},\ \sum_{j} C_{j})$.}

\takeaway{Pequenas perturba\c{c}\~oes nas chaves causam pequenas mudan\c{c}as
na ordem: o espa\c{c}o de permuta\c{c}\~oes \'e explorado \textbf{sem
operadores especializados}.}

\end{frame}

popula\c{c}\~oes com migra\c{c}\~ao, \emph{implicit path-relinking},
\emph{shaking} e \emph{reset} parcial.}

\takeaway{O decodificador \textbf{sempre} produz escalonamento vi\'avel:
sem reparo nem penalidade.}

\end{frame}

\section{Conclusão}

%----------------------------------------------------------------------------------------
% 13. CONCLUSÕES
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Conclus\~oes}

\begin{itemize}\setlength{\itemsep}{6pt}
\item Formulamos o \textbf{MOTMDSP}, extens\~ao biobjetivo do TMDSP que trata
      explicitamente o compromisso entre \emph{makespan} e tempo total de
      conclus\~ao.

\item \textbf{Estendemos o decodificador} de Andrade et al. (2019) para os dois
      objetivos, o que permitiu aplicar ao problema tanto o \alg{NS-BRKGA}
      quanto os cinco algoritmos de refer\^encia.

\item Em $96$ inst\^ancias de larga escala, o \alg{NS-BRKGA} obteve as
      \textbf{melhores m\'edias} de $\HVR$ e $\NIGDplus$, com vantagem mais
      n\'itida nas inst\^ancias \texttt{rare}.

\item O desempenho \textbf{se mant\'em} \`a medida que o n\'umero de tarefas ou
      de m\'aquinas cresce, sob o mesmo or\c{c}amento de tempo.

\item \textbf{Uso pr\'atico:} em vez de um \'unico escalonamento, o operador
      recebe um \textbf{conjunto de compromissos} entre terminar a campanha
      mais cedo e concluir mais ve\'iculos antes.
\end{itemize}

\end{frame}
