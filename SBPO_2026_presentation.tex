%----------------------------------------------------------------------------------------
% 3. DO FOTA AO TMDSP
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Do FOTA ao TMDSP}

{\small O FOTA \'e modelado pelo \emph{Time- and Machine-Dependent Scheduling
Problem} (TMDSP), de Andrade et al.\ (2019):}

\smallskip

% Sobrescreve, s\'o neste frame, o \normalsize que o pre\^ambulo fixa
% para o corpo das listas.
\setbeamertemplate{itemize/enumerate body begin}{\footnotesize}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\footnotesize
\textbf{Dados}
\begin{itemize}\setlength{\itemsep}{2pt}
\item<1-> $J$ --- \textbf{tarefas}: ve\'iculos, com demanda $d_{j}$.
\item<1-> $M$ --- \textbf{m\'aquinas}: setores da rede.
\item<1-> $T$ --- \textbf{per\'iodos} do horizonte.
\item<2-> $c_{mt}$, $\eta_{mt}$ --- capacidade e taxa da m\'aquina,
      \alert{vari\'aveis no tempo}.
\item<3-> $q_{jmt}$ --- capacidade que $j$ consome em $m$ no per\'iodo $t$;
      \alert{s\'o existe} se o ve\'iculo estiver sob aquele setor ali.
\end{itemize}

\column{0.48\textwidth}
\footnotesize
\textbf{Regras}
\begin{itemize}\setlength{\itemsep}{2pt}
\item<4-> Cada tarefa usa \textbf{uma m\'aquina por per\'iodo}, no m\'aximo;
\item<4-> \'e \textbf{preempt\'avel entre per\'iodos} e fica ativa at\'e
      atender $d_{j}$;
\item<4-> $C_{j}$ \'e o \textbf{\'ultimo per\'iodo ativo} de $j$.
\item<5-> Por per\'iodo: $r_{\max}$ limita a vaz\~ao total da rede e
      $k_{\max}$, o n\'umero de tarefas ativas.
\end{itemize}
\end{columns}

\vspace{0.1em}

\onslide<6->{\takeaway{Objetivo original, mono-objetivo:
$\displaystyle \min \sum_{j \in J} C_{j}$ --- o tempo total de conclus\~ao.}}

\end{frame}

%----------------------------------------------------------------------------------------
% 4. POR QUE MULTIOBJETIVO
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Minimizar $\sum_{j} C_{j}$ n\~ao controla o \emph{makespan}}

\begin{columns}[c,onlytextwidth]
\column{0.47\textwidth}

\small
A correla\c{c}\~ao entre os dois crit\'erios \'e \textbf{fraca}: um
escalonamento \'otimo em $\sum_{j} C_{j}$ pode deixar algumas tarefas
terminando \textbf{muito depois} das demais.

\medskip

Propomos o \alert{MOTMDSP}, que minimiza \textbf{simultaneamente}:

\smallskip

\begin{center}
\small
$\displaystyle \min\ C_{\max} = \max_{j \in J} C_{j}$
\;\; (\emph{makespan})

\smallskip

$\displaystyle \min \sum_{j \in J} C_{j}$
\;\; (tempo total de conclus\~ao)
\end{center}

\smallskip

{\scriptsize\color{black!55} Observado por Andrade et al.\ (2019).}

\column{0.49\textwidth}
\centering
\begin{tikzpicture}[scale=0.72, font=\small]
\draw[-Stealth, thick] (0,0) -- (7.4,0) node[below left=2pt and 0pt] {$C_{\max}$};
\draw[-Stealth, thick] (0,0) -- (0,4.9) node[above right=0pt and -6pt] {$\sum_{j} C_{j}$};
\draw[ultra thick, mydarkblue1, smooth, tension=0.7]
    plot coordinates {(1.0,4.2) (1.7,3.1) (2.6,2.4) (3.7,2.0) (5.0,1.8) (6.4,1.7)};
\foreach \x/\y in {1.0/4.2, 1.7/3.1, 2.6/2.4, 3.7/2.0, 5.0/1.8, 6.4/1.7} {
    \fill[mydarkblue1] (\x,\y) circle (3pt);
}
\node[mydarkblue1, font=\small\bfseries, anchor=west] at (3.4,3.6)
  {fronteira de Pareto};
\node[mydarkorange1, font=\scriptsize, align=center, anchor=south west]
  at (0.55,0.35)
  {nenhuma solu\c{c}\~ao \'e melhor\\nos \textbf{dois} objetivos};
\end{tikzpicture}
\end{columns}

\end{frame}

%----------------------------------------------------------------------------------------
% 5. FORMULAÇÃO DO MOTMDSP
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{MOTMDSP: formula\c{c}\~ao}

\begin{center}
{\large $\displaystyle \min\ C_{\max}
 \qquad\text{e}\qquad
 \min \sum_{j \in J} C_{j}$}
\end{center}

{\small sujeito a, para todo per\'iodo $t$ (com $\hat{t} = t \bmod |T|$):}

\smallskip

{\footnotesize
\begin{tabular}{@{} r @{\;\;} l @{}}
$\displaystyle\sum_{m \in M} x_{jmt} \le 1$
  & uma m\'aquina por per\'iodo, para cada tarefa $j$ \\[0.85em]
$\displaystyle\sum_{j \in J} y_{jmt} \le c_{m\hat{t}}$
  & capacidade da m\'aquina $m$ \\[0.85em]
$\displaystyle\sum_{j \in J}\sum_{m \in M} \eta_{m\hat{t}}\, y_{jmt} \le r_{\max}$
  & vaz\~ao total da rede \\[0.85em]
$\displaystyle\sum_{j \in J} a_{jt} \le k_{\max}$
  & tarefas ativas simultaneamente \\[0.85em]
$\displaystyle\sum_{m \in M}\sum_{t} \eta_{m\hat{t}}\, y_{jmt} = d_{j}$
  & demanda da tarefa $j$ atendida \\[0.85em]
$C_{j} \ge t\,a_{jt}, \quad C_{\max} \ge C_{j}$
  & tempos de conclus\~ao \\
\end{tabular}}

\medskip

\srcnote{$x_{jmt}$: atribui\c{c}\~ao; $y_{jmt}$: capacidade usada;
$a_{jt}$: tarefa ativa. As restri\c{c}\~oes de in\'icio, propaga\c{c}\~ao de
atividade e t\'ermino seguem Andrade et al. (2019).}

\end{frame}

\begin{frame}
\frametitle{Chaves aleat\'orias definem prioridades de escalonamento}

Um cromossomo \'e um vetor com \textbf{uma chave por tarefa}:
\[
\mathbf{v} = (v_1, \ldots, v_{|J|}), \qquad v_j \in [0,\,1).
\]

\textbf{Decodifica\c{c}\~ao:} ordenar as tarefas por valor \textbf{decrescente}
da chave $\Rightarrow$ ordem de prioridade.

\bigskip

\begin{center}
\begin{tikzpicture}[>=Stealth, font=\small, x=1cm, y=1cm]
% vetor original
\node[font=\small, anchor=east] at (-0.52,0) {$\mathbf{v}$:};
\foreach \val [count=\i from 0] in
  {{$0{,}6$}, {$0{,}9$}, {$0{,}8$}, {$0{,}5$}, {$0{,}3$}, {$0{,}2$}} {
    \node[draw, minimum width=0.86cm, minimum height=0.66cm] at (\i*0.88, 0) {\val};
    \node[font=\scriptsize, above=0pt] at (\i*0.88, 0.33)
      {$j_{\the\numexpr\i+1}$};
}
% seta
\draw[-Stealth, ultra thick, mydarkorange1] (4.95,0) -- (6.15,0);
\node[font=\scriptsize, align=center, below=1pt] at (5.55,-0.10)
  {ordem\\decrescente};
% vetor ordenado
\foreach \job/\col/\txt [count=\i from 0] in
  {{$j_2$}/mylightorange1/black, {$j_3$}/mylightblue1/black,
   {$j_1$}/white/black, {$j_4$}/white/black,
   {$j_5$}/white/black, {$j_6$}/white/black} {
    \node[draw, fill=\col, text=\txt, minimum width=0.68cm, minimum height=0.66cm]
        at (6.55+\i*0.72, 0) {\job};
}
\node[font=\scriptsize, above=0pt] at (8.35,0.33) {\textbf{prioridade}};
\end{tikzpicture}
\end{center}

\bigskip

\takeaway{Pequenas perturba\c{c}\~oes nas chaves causam pequenas mudan\c{c}as na
ordem: o espa\c{c}o de permuta\c{c}\~oes \'e explorado \textbf{sem operadores
especializados}.}

\end{frame}

\section{Conclusão}

%----------------------------------------------------------------------------------------
% 13. CONCLUSÕES
%----------------------------------------------------------------------------------------
\begin{frame}
\frametitle{Conclus\~oes}

\begin{itemize}\setlength{\itemsep}{6pt}
\item Formulamos o \textbf{MOTMDSP}, extens\~ao biobjetivo do TMDSP que trata
      explicitamente o compromisso entre \emph{makespan} e tempo total de
      conclus\~ao.

\item \textbf{Estendemos o decodificador} de Andrade et al. (2019) para os dois
      objetivos, o que permitiu aplicar ao problema tanto o \alg{NS-BRKGA}
      quanto os cinco algoritmos de refer\^encia.

\item Em $96$ inst\^ancias de larga escala, o \alg{NS-BRKGA} obteve as
      \textbf{melhores m\'edias} de $\HVR$ e $\NIGDplus$, com vantagem mais
      n\'itida nas inst\^ancias \texttt{rare}.

\item O desempenho \textbf{se mant\'em} \`a medida que o n\'umero de tarefas ou
      de m\'aquinas cresce, sob o mesmo or\c{c}amento de tempo.

\item \textbf{Uso pr\'atico:} em vez de um \'unico escalonamento, o operador
      recebe um \textbf{conjunto de compromissos} entre terminar a campanha
      mais cedo e concluir mais ve\'iculos antes.
\end{itemize}

\end{frame}
