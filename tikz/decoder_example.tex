% ---------------------------------------------------------------------------
% Decodificador: chaves aleatórias -> ordem de prioridade -> alocação gulosa
% nos intervalos máquina-período.
% Adaptado de decoding_example.tikz, de Andrade et al. (2019): cores da
% paleta PuOr, rótulos em português e marcação dos tempos de conclusão.
% ---------------------------------------------------------------------------
\begin{tikzpicture}[x=5mm, y=5mm, font=\sffamily]

  \def\basevetorX{6};
  \def\basevetorY{12};
  \def\distancia{22};

  % Vetor de chaves
  \draw(\basevetorX, \basevetorY) grid[step=2] +(12, 2);

  \foreach \i/\v in {1/0{,}6, 2/0{,}9, 3/0{,}8, 4/0{,}5, 5/0{,}3, 6/0{,}2} {
    \node(Bid\i) at (2*\i-1 + \basevetorX, \basevetorY-0.5) {\large $j_{\i}$};
    \node at (2*\i + \basevetorX - 1, \basevetorY+1) {\large $\v$};
  }
  \node[anchor=east] at (\basevetorX-.5, \basevetorY+1) {\Large Vetor $v$};
  \node[anchor=east]  at (\basevetorX-.5, \basevetorY-0.5) {\Large Tarefas};

  % Vetor ordenado
  \draw (\basevetorX+\distancia,\basevetorY) grid[step=2] +(12, 2);

  \foreach \j/\i/\v in {1/2/0{,}9, 2/3/0{,}8, 3/1/0{,}6, 4/4/0{,}5, 5/5/0{,}3, 6/6/0{,}2}{
    \node(SJob\i) at (2*\j-1 + \basevetorX + \distancia, \basevetorY-0.5) {\large $j_{\i}$};
    \node at (\distancia+\basevetorX+ 2 * \j - 1, \basevetorY+1) {\large $\v$};
  }

  \tikzset{
    SetaSort/.style={dashed, color=black, ultra thick, ->}
  }

  \draw[SetaSort] (\basevetorX+12, \basevetorY+1) --
    (\basevetorX+\distancia-0.25, \basevetorY+1);

  \node[fill=white, inner sep=1pt] at (23, 13.8) {\Large Ordenar por};
  \node[fill=white, inner sep=1pt] at (23, 12.3) {\Large chave decrescente};

  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
  %% Separadores de período
  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

  \node at (5, 2) {\Large\underline{Per\'iodo $t_1$}};

  \draw[thick, color=gray] (11, 1) -- (11, 7.5);
  \node at (17, 2) {\Large\underline{Per\'iodo $t_2$}};

  \draw[thick, color=gray] (23, 1) -- (23, 7.5);
  \node at (29, 2) {\Large\underline{Per\'iodo $t_3$}};

  \draw[thick, color=gray] (35, 1) -- (35, 7.5);
  \node at (41, 2) {\Large\underline{Per\'iodo $t_4$}};

  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
  %% Período 1
  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

  \path (0, 3) pic {MachineCap={lenght=10}} node(Per1Mach2) {};
  \node[anchor=west] at (0, 4.5) {\large\itshape M\'aquina $m_2$};

  \path (0, 5) pic {MachineCap={lenght=10}} node(Per1Mach1) {};
  \node[anchor=west] at (0, 6.5) {\large\itshape M\'aquina $m_1$};

  \path (0.05, 5.05) pic {Machine={lenght=10, color=JobALight, label=1}};
  \path (0.05, 3.05) pic {Machine={lenght=3, color=JobALight, label=1}};

  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
  %% Período 2
  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

  \path (12, 3) pic {MachineCap={lenght=10}} node(Per2Mach2) {};
  \node[anchor=west] at (12, 4.5) {\large\itshape M\'aquina $m_2$};

  \path (12.05, 3.05) pic {Machine={lenght=2, color=JobALight, label=1}};
  \path (14.05, 3.05) pic {Machine={lenght=8, color=JobBLight, label=1}};

  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
  %% Período 3
  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

  \path (24, 3) pic {MachineCap={lenght=10}} node(Per3Mach2) {};
  \node[anchor=west] at (24, 4.5) {\large\itshape M\'aquina $m_2$};

  \path (24, 5) pic {MachineCap={lenght=10}} node(Per3Mach1) {};
  \node[anchor=west] at (24, 6.5) {\large\itshape M\'aquina $m_1$};

  \path (24.05, 5.05) pic {Machine={lenght=4, color=JobBLight, label=1}};
  \path (24.05, 3.05) pic {Machine={lenght=4, color=JobBLight, label=1}};

  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
  %% Período 4
  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

  \path (36, 3) pic {MachineCap={lenght=10}} node(Per4Mach2) {};
  \node[anchor=west] at (36, 4.5) {\large\itshape M\'aquina $m_2$};

  \path (36, 5) pic {MachineCap={lenght=10}} node(Per4Mach1) {};
  \node[anchor=west] at (36, 6.5) {\large\itshape M\'aquina $m_1$};

  \path (36.05, 5.05) pic {Machine={lenght=2, color=JobBLight, label=1}};
  \path (36.05, 3.05) pic {Machine={lenght=2, color=JobBLight, label=1}};

  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
  %% Setas da tarefa j2 (primeira na ordem de prioridade)
  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

  \draw[inner sep=0mm, minimum size=4mm, thick, fill, rounded corners=0.5mm, JobALight]
  (28.25, 11) rectangle (29.75, 11.9)
  node[color=black, fill opacity=1.0, pos=0.5] {\bfseries\large $j_2$};

  \draw[->,ultra thick, rounded corners=2mm, JobADark]
  (SJob2) -- ++(0, -2) -- ++(-24, 0) -- ++(0, -3.25);

  \draw[->,ultra thick, rounded corners=2mm, JobADark]
  (11, 9.5) -- ++(0, -6) -- ++(-0.8, 0);

  \draw[->,ultra thick, rounded corners=2mm, JobADark]
  (11, 9.5) -- ++(0, -6) -- ++(0.9, 0);

  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
  %% Setas da tarefa j3 (segunda na ordem de prioridade)
  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

  \draw[inner sep=0mm, minimum size=4mm, thick, fill, rounded corners=0.5mm, JobBLight]
  (30.25, 11) rectangle (31.75, 11.9)
  node[color=black, fill opacity=1.0, pos=0.5] {\bfseries\large $j_3$};

  \draw[ultra thick, JobBDark]
  (SJob3) -- ++(0, -3.5);

  \draw[->,ultra thick, rounded corners=2mm, JobBDark]
  ($(SJob3) + (0, -3.5)$) -- (18, 8) -- ++(0, -3.8);

  \draw[->,ultra thick, rounded corners=2mm, JobBDark]
  ($(SJob3) + (0, -3.5)$) -- (31, 3.5) -- ++(-3, 0);

  \draw[->,ultra thick, rounded corners=2mm, JobBDark]
  ($(SJob3) + (0, -3.5)$) -- (31, 3.5) -- ++(4.9, 0);

  \draw[<->,ultra thick, rounded corners=2mm, JobBDark]
  (28, 5.5) -- (35.9, 5.5);

  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
  %% Tempos de conclusão das duas tarefas alocadas
  %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

  \draw[JobADark, ultra thick] (22.6, 2.6) -- (22.6, 0.4);
  \node[JobADark, anchor=east, font=\large] at (22.4, 0.0) {$C_{j_2} = 2$};

  \draw[JobBDark, ultra thick] (46.4, 2.6) -- (46.4, 0.4);
  \node[JobBDark, anchor=east, font=\large] at (46.2, 0.0) {$C_{j_3} = 4$};

\end{tikzpicture}
